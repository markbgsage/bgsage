cmake_minimum_required(VERSION 3.18)
project(bgsage_mobile LANGUAGES CXX)
option(BGSAGE_MOBILE_REFERENCE_PYTHON "Also build the Python extension for local parity checks" OFF)
set(BGBOT_BUILD_PYTHON ${BGSAGE_MOBILE_REFERENCE_PYTHON} CACHE BOOL "" FORCE)
set(BGBOT_BUILD_APPS OFF CACHE BOOL "" FORCE)
include(${CMAKE_CURRENT_LIST_DIR}/../cpp/CMakeLists_cpu.txt)
add_library(bgsage_mobile SHARED ../cpp/src/mobile.cpp)
target_link_libraries(bgsage_mobile PRIVATE bgbot_lib)
target_compile_definitions(bgsage_mobile PRIVATE NOMINMAX)
set_target_properties(bgsage_mobile PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
